**Script structure:**
```typescript
import "dotenv/config";
import { drizzle } from "drizzle-orm/node-postgres";
import { Pool } from "pg";
import argon2 from "argon2";
import * as schema from "./schema";

const pool = new Pool({ connectionString: process.env.DATABASE_URL! });
const db = drizzle(pool, { schema });

async function seed() {
  console.log("Seeding database...");

  // 1. Branches
  const insertedBranches = await db.insert(schema.branches).values([...]).onConflictDoNothing().returning();
  // Use insertedBranches[0].id for BRN01, insertedBranches[1].id for BRN02
  // NOTE: onConflictDoNothing returns empty array on conflict — handle this:
  // If insertedBranches is empty (re-run), query existing branches
  const brn01 = insertedBranches.find(b => b.code === "BRN01")
    ?? (await db.select().from(schema.branches).where(...).limit(1))[0];

  // 2. Hash passwords (do this early, they take time)
  const [adminHash, pharmHash, cashHash, invHash] = await Promise.all([
    argon2.hash("Admin@123"),
    argon2.hash("Pharm@123"),
    argon2.hash("Cash@123"),
    argon2.hash("Inv@123"),
  ]);

  // 3. Users
  await db.insert(schema.users).values([...]).onConflictDoNothing();

  // 4. Medicine categories
  const insertedCats = await db.insert(schema.medicineCategories).values([...]).onConflictDoNothing().returning();
  // Handle re-run: if empty, query existing
  
  // 5. Medicines
  await db.insert(schema.medicines).values([...]).onConflictDoNothing();

  // 6. Suppliers
  await db.insert(schema.suppliers).values([...]).onConflictDoNothing();

  console.log("Seed complete.");
}

seed()
  .then(() => { pool.end(); process.exit(0); })
  .catch((e) => { console.error(e); pool.end(); process.exit(1); });
```
